# fgt-a — Cartio lab, site A edge
# FortiOS 7.6 CLI. Paste section by section, in order (each section only references objects created
# above it). Lines starting with # are comments.
#
# Role: site A edge. port1 = WAN to isp (eBGP AS 65010 to AS 64500, default route learned),
# port2 = LAN to core-a (OSPF area 0, default originated), port3 = reserved FortiLink port for a
# physical FortiSwitch or FortiAP bridged in over pnet2. LLDP on port2. Syslog to srv-a.
# Exercises the FortiGate driver (REST API and SSH fallback) and, in phase 3, the ADR-0009
# deep-inspection run (policy "coll-a-deep-inspection", disabled until then).
# The read-only SSH admin, the API key and the SNMPv3 user are added after first boot
# (labs/cartio/README.md), so no collector credential lives in this file.
#

#### system / management
config system global
    set hostname "fgt-a"
    set admintimeout 30
    set admin-lockout-threshold 5
    set admin-lockout-duration 300
    set strong-crypto enable
    set admin-https-redirect enable
    set lldp-transmission enable
    set lldp-reception enable
end
config system dns
    set primary 1.1.1.1
    set secondary 1.0.0.1
end
config system ntp
    set ntpsync enable
    set type custom
    config ntpserver
        edit 1
            set server "198.51.100.254"
        next
    end
end

#### interfaces
config system interface
    edit "port1"
        set mode static
        set ip 198.51.100.2 255.255.255.252
        set allowaccess ping https ssh
        set role wan
        set alias "to-isp"
        set description "to isp Gi0/1 (eBGP 198.51.100.1)"
    next
    edit "port2"
        set mode static
        set ip 172.30.0.1 255.255.255.252
        set allowaccess ping https ssh snmp
        set role lan
        set alias "to-core-a"
        set description "to core-a Gi1 (OSPF area 0)"
        set lldp-reception enable
        set lldp-transmission enable
    next
    edit "port3"
        set alias "fortilink-reserved"
        set description "reserved for a physical FortiSwitch or FortiAP over pnet2"
        set status down
    next
end

#### read-only access for the collector (ADR-0003, ADR-0004)
config system accprofile
    edit "cartio-ro"
        set secfabgrp read
        set ftviewgrp read
        set authgrp read
        set sysgrp read
        set netgrp read
        set loggrp read
        set fwgrp read
        set vpngrp read
        set utmgrp read
        set wanoptgrp read
        set wifi read
    next
end
config system api-user
    edit "cartio-ro-api"
        set comments "Cartio collector, read-only"
        set accprofile "cartio-ro"
        set vdom "root"
        config trusthost
            edit 1
                set ipv4-trusthost 172.30.20.0 255.255.255.0
            next
            edit 2
                set ipv4-trusthost 10.100.51.0 255.255.255.0
            next
        end
    next
end

#### monitoring (passing assessment checks: NTP above, syslog and SNMP here; SNMPv3 user after boot)
config system snmp sysinfo
    set status enable
    set description "Cartio lab site A edge"
    set contact-info "cartio-lab"
    set location "site-a"
end
config log syslogd setting
    set status enable
    set server "172.30.20.21"
end

#### addresses
config firewall address
    edit "lab-172-30"
        set subnet 172.30.0.0 255.255.0.0
    next
    edit "mgmt-lan"
        set comment "EVE management LAN (pnet0): engineer laptop, dev controller"
        set subnet 10.100.51.0 255.255.255.0
    next
    edit "coll-a"
        set subnet 172.30.20.50 255.255.255.255
    next
end

#### policies
# NAT is off: isp NATs the lab to the management LAN, and leaves traffic to it untranslated.
config firewall policy
    edit 1
        set name "coll-a-deep-inspection"
        set comments "ADR-0009 manual run: enable for the test, it already sits above lan-to-wan"
        set status disable
        set srcintf "port2"
        set dstintf "port1"
        set action accept
        set srcaddr "coll-a"
        set dstaddr "all"
        set schedule "always"
        set service "HTTPS"
        set utm-status enable
        set ssl-ssh-profile "deep-inspection"
        set application-list "default"
        set logtraffic all
    next
    edit 2
        set name "lan-to-wan"
        set srcintf "port2"
        set dstintf "port1"
        set action accept
        set srcaddr "lab-172-30"
        set dstaddr "all"
        set schedule "always"
        set service "ALL"
        set ssl-ssh-profile "certificate-inspection"
        set logtraffic all
    next
    edit 3
        set name "mgmt-lan-to-lab"
        set srcintf "port1"
        set dstintf "port2"
        set action accept
        set srcaddr "mgmt-lan"
        set dstaddr "lab-172-30"
        set schedule "always"
        set service "PING" "SSH" "HTTPS" "SNMP"
        set logtraffic all
    next
end

#### routing
config router ospf
    set router-id 172.30.0.1
    set default-information-originate enable
    config area
        edit 0.0.0.0
        next
    end
    config ospf-interface
        edit "port2"
            set interface "port2"
            set network-type point-to-point
        next
    end
    config network
        edit 1
            set prefix 172.30.0.0 255.255.255.252
        next
    end
end
config router bgp
    set as 65010
    set router-id 172.30.0.1
    config neighbor
        edit "198.51.100.1"
            set remote-as 64500
            set description "isp"
        next
    end
    config redistribute "connected"
        set status enable
    end
    config redistribute "ospf"
        set status enable
    end
end

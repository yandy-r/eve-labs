# b1-fgt1 — SD-WAN spoke
# FortiOS 7.6.8 CLI. Paste section by section, in order (each section only references objects created
# above it). Lines starting with # are comments.
#
# Role: SD-WAN spoke — 2 ISP VLANs (EVE-INET-201, EVE-INET-202) tagged on port1/port3 (pnet0 trunk),
# 4 IP-less overlays (2 hubs x 2 ISPs), BGP on loopback 10.200.0.51 to both hubs (iBGP AS 64001), ADVPN 2.0
# (advpn-select on zones HUB1/HUB2), eBGP + default-originate to the LAN router(s), local internet breakout.
# DC-APPS uses tie-break fib-best-match so host routes (/32s from the BGWs) pick the hub of the site owning the host.
# INET VLANs: static addressing; reserve .51 on EVE-INET-202 (it has a DHCP pool). PSK is a lab placeholder.
#

#### SD-WAN
config system sdwan
    set status enable
    config zone
        edit "underlay"
        next
        edit "HUB1"
        next
        edit "HUB2"
        next
    end
    config members
        edit 1
            set interface "wan1"
            set zone "underlay"
            set gateway 10.100.201.1
        next
        edit 2
            set interface "wan2"
            set zone "underlay"
            set gateway 10.100.202.1
        next
        edit 11
            set interface "H1_INET1"
            set zone "HUB1"
            set source 10.200.0.51
        next
        edit 12
            set interface "H1_INET2"
            set zone "HUB1"
            set source 10.200.0.51
        next
        edit 21
            set interface "H2_INET1"
            set zone "HUB2"
            set source 10.200.0.51
        next
        edit 22
            set interface "H2_INET2"
            set zone "HUB2"
            set source 10.200.0.51
        next
    end
    config health-check
        edit "HUB_HC"
            set server "10.200.255.251"
            set protocol ping
            set update-cascade-interface disable
            set update-static-route disable
            set members 11 12 21 22
            config sla
                edit 1
                    set latency-threshold 150
                    set jitter-threshold 30
                    set packetloss-threshold 2
                next
            end
        next
        edit "INET_HC"
            set server "1.1.1.1" "8.8.8.8"
            set protocol ping
            set members 1 2
            config sla
                edit 1
                    set latency-threshold 150
                    set jitter-threshold 30
                    set packetloss-threshold 2
                next
            end
        next
    end
    config zone
        edit "HUB1"
            set advpn-select enable
            set advpn-health-check "HUB_HC"
        next
        edit "HUB2"
            set advpn-select enable
            set advpn-health-check "HUB_HC"
        next
    end
    config service
        edit 1
            set name "DC-APPS"
            set mode sla
            set dst "DC-NETS"
            set src "all"
            set tie-break fib-best-match
            config sla
                edit "HUB_HC"
                    set id 1
                next
            end
            set priority-members 11 12 21 22
        next
        edit 2
            set name "CORP-OVERLAY"
            set mode sla
            set dst "CORP-NETS"
            set src "all"
            config sla
                edit "HUB_HC"
                    set id 1
                next
            end
            set priority-members 11 12 21 22
        next
        edit 3
            set name "INTERNET-DIA"
            set mode sla
            set dst "all"
            set src "all"
            config sla
                edit "INET_HC"
                    set id 1
                next
            end
            set priority-members 1 2
        next
    end
end

# Synthetic: VLAN links, nmcli, netplan and interfaces heredocs (EVE-MGMT-203 uplink)
ip link add link eth0 name eth0.1240 type vlan id 1240
ip addr add 172.31.240.10/30 dev eth0.1240
ip route add 10.100.0.0/16 via 172.31.240.9 dev eth0.1240
vconfig add eth1 203
ifconfig eth1.203 10.100.51.50/24
route add -net 10.200.0.0 netmask 255.255.0.0 gw 10.100.51.1
route add -net 10.201.0.0/16 gw 10.100.51.1

echo "nameserver 9.9.9.9" >> /etc/resolv.conf
echo "169.1.100.16 dc2-srv16" >> /etc/hosts

  ethernets:
    eth3:
      addresses: [10.100.53.50/24]
      routes:
        - to: 10.0.0.0/8
          via: 10.100.53.1
      nameservers:
        addresses: [8.8.8.8, 8.8.4.4]
        search: [netlab.local, yandy.dev]
EOF
cat << 'EOF' > /etc/network/interfaces
auto eth4
iface eth4 inet static
    address 10.100.54.50
    netmask 255.255.255.0
    gateway 10.100.54.1
    dns-nameservers 1.1.1.1 1.0.0.1
